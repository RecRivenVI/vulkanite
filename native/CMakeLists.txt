cmake_minimum_required(VERSION 3.24)
project(vulkanite_dlss LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(JNI REQUIRED)
find_package(Vulkan REQUIRED)
set(NGX_SDK "" CACHE PATH "NVIDIA DLSS SDK root (tested commit 374959484e79a640feaba44c93ac8cfb0a03f5b5)")
if(NOT WIN32 OR NOT EXISTS "${NGX_SDK}/include/nvsdk_ngx_vk.h")
    message(FATAL_ERROR "Provide -DNGX_SDK=<official SDK>; this first bridge targets Windows x64")
endif()
add_library(vulkanite_dlss SHARED dlss_bridge.cpp)
target_include_directories(vulkanite_dlss PRIVATE ${JNI_INCLUDE_DIRS} "${NGX_SDK}/include")
target_link_libraries(vulkanite_dlss PRIVATE Vulkan::Vulkan "${NGX_SDK}/lib/Windows_x86_64/x64/nvsdk_ngx_d.lib")
target_compile_definitions(vulkanite_dlss PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
